// LISTA DE CIDADES PERMITIDAS (SÓ ESTAS VALEM) const CIDADES_PERMITIDAS = [ "Xinguara", "Agua Azul do Norte", "Água Azul do Norte", "Tucumã", "Tucuma", "Ourilândia do Norte", "Ourilandia do Norte", "São Felix do Xingu", "Redenção", "Redencao", "Rio Maria", "Sapucaia", "Sapucai", "Canaã dos Carajás", "Canaa dos Carajas" ]; async function enviar() { if(localStorage.getItem(LOCK_KEY)) return alert("Voto já registrado."); if(!sel) return alert("Selecione um candidato."); const btn = document.getElementById("btn-principal"); btn.disabled = true; try { const r = await fetch('https://ipapi.co/json/'); const d = await r.json(); const cidadeDetectada = d.city || ""; // VALIDAÇÃO DE REGIAO NA HORA DO VOTO const ehValida = CIDADES_PERMITIDAS.some(c => cidadeDetectada.toLowerCase().includes(c.toLowerCase())); if(!ehValida) { alert("Acesso Negado: Esta enquete é restrita aos moradores da região Sul do Pará (Xinguara, Água Azul, etc)."); btn.disabled = false; return; } const ip = d.ip.replace(/\./g, '_'); db.ref("ips/" + ip).once("value", s => { if(s.exists()) { alert("Este IP já votou."); localStorage.setItem(LOCK_KEY, "true"); return mostrar(); } const infoLocal = cidadeDetectada + " (" + (d.region_code || "PA") + ")"; const uid = Date.now(); db.ref("voters/" + uid).set({c:infoLocal, can:sel, t:new Date().toLocaleString('pt-BR')}); db.ref("ips/" + ip).set(true); db.ref("votes/" + sel).transaction(v => (v||0)+1); localStorage.setItem(LOCK_KEY, "true"); mostrar(); }); } catch(e){ alert("Erro de conexão."); btn.disabled = false; } } // FUNÇÃO DE SINCRONIZAÇÃO NO PAINEL ADM function sync() { db.ref("voters").once("value", s => { let vtr = s.val() || {}; // Reinicia contagem para auditoria (Eraclito mantém os 71 base) let contagemAuditada = { cicero:0, renan:0, polaco:0, eraclito:71, romes:0, weverson:0, yan:0, joão:0 }; Object.values(vtr).forEach(v => { let cid = (v.c || "").split(" (")[0]; // Pega só o nome da cidade let canId = v.can; // Verifica se a cidade do voto está na sua lista permitida let ehCidadeReal = CIDADES_PERMITIDAS.some(c => cid.toLowerCase().includes(c.toLowerCase())); if (ehCidadeReal) { if (contagemAuditada.hasOwnProperty(canId)) { contagemAuditada[canId]++; } } }); if(confirm("Deseja aplicar o FILTRO REGIONAL? Apenas votos das cidades selecionadas (Xinguara, Água Azul, etc) serão mantidos.")) { Object.keys(contagemAuditada).forEach(id => { db.ref("votes/" + id).set(contagemAuditada[id]); }); alert("Sucesso! Placar regionalizado e robôs eliminados."); location.reload(); } }); }
